## compile
docker run --rm   -v "$PWD:/src"   -w /src   mcr.microsoft.com/dotnet/sdk:10.0   dotnet build -v minimal

## Install
cd /appdata/jellyfin-plugin-dev

mkdir -p /appdata/jellyfin-plugin-staging/PerLibraryPlaybackThresholds_0.1.5.0

cp bin/Debug/net10.0/PerLibraryPlaybackThresholds.dll \
  /appdata/jellyfin-plugin-staging/PerLibraryPlaybackThresholds_0.1.5.0/

chown -R 1000:1000 /appdata/jellyfin-plugin-staging/PerLibraryPlaybackThresholds_0.1.5.0/

docker exec jellyfin mkdir -p \
  /config/plugins/PerLibraryPlaybackThresholds_0.1.5.0

docker cp \
  /appdata/jellyfin-plugin-staging/PerLibraryPlaybackThresholds_0.1.5.0/PerLibraryPlaybackThresholds.dll \
  jellyfin:/config/plugins/PerLibraryPlaybackThresholds_0.1.5.0/

docker restart jellyfin
